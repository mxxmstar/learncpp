### 2. 实现核心基类

#### 文件: `modules/web/include/web/core/iservice.h`

```cpp
#pragma once

#include <string>
#include <memory>

namespace web {
namespace core {

/**
 * @brief 服务基类接口
 * 
 * 所有服务必须实现此接口
 */
class IService {
public:
    virtual ~IService() = default;
    
    /// @brief 获取服务名称（唯一标识）
    virtual std::string name() const = 0;
    
    /// @brief 初始化服务（加载配置、分配资源）
    virtual bool initialize() = 0;
    
    /// @brief 启动服务（开始处理业务）
    virtual bool start() = 0;
    
    /// @brief 停止服务（释放资源、清理状态）
    virtual void stop() = 0;
    
    /// @brief 检查服务是否正在运行
    virtual bool isRunning() const = 0;
    
    /// @brief 健康检查
    virtual bool isHealthy() const { return isRunning(); }
};

using ServicePtr = std::shared_ptr<IService>;

} // namespace core
} // namespace web
```

#### 文件: `modules/web/include/web/core/service_manager.h`

```cpp
#pragma once

#include "iservice.h"
#include <vector>
#include <map>
#include <memory>
#include <functional>

namespace web {
namespace core {

/**
 * @brief 服务管理器
 * 
 * 负责服务的注册、依赖管理、启动/停止顺序控制
 */
class ServiceManager {
public:
    static ServiceManager& getInstance();
    
    // 禁止拷贝
    ServiceManager(const ServiceManager&) = delete;
    ServiceManager& operator=(const ServiceManager&) = delete;
    
    /// @brief 注册服务
    bool registerService(ServicePtr service);
    
    /// @brief 获取服务
    template<typename T>
    std::shared_ptr<T> getService(const std::string& name) {
        auto it = services_.find(name);
        if (it != services_.end()) {
            return std::dynamic_pointer_cast<T>(it->second);
        }
        return nullptr;
    }
    
    /// @brief 初始化所有服务
    bool initializeAll();
    
    /// @brief 启动所有服务（按依赖顺序）
    bool startAll();
    
    /// @brief 停止所有服务（逆序）
    void stopAll();
    
    /// @brief 获取所有服务状态
    struct ServiceStatus {
        std::string name;
        bool running;
        bool healthy;
    };
    std::vector<ServiceStatus> getAllStatus() const;
    
private:
    ServiceManager() = default;
    ~ServiceManager() = default;
    
    std::map<std::string, ServicePtr> services_;
    std::vector<std::string> startup_order_;  // 启动顺序
};

} // namespace core
} // namespace web
```

#include "../core/iservice.h"

namespace web {
namespace services {

/**
 * @brief 视频流水线服务
 * 
 * 封装 VideoPipelineManager，提供多路视频流管理能力
 */
class VideoPipelineService : public core::IService {
public:
    VideoPipelineService();
    ~VideoPipelineService() override;
    
    // IService 接口实现
    std::string name() const override { return "video_pipeline"; }
    bool initialize() override;
    bool start() override;
    void stop() override;
    bool isRunning() const override { return running_; }
    
    // 业务方法
    int createPipeline(const PipelineConfig& config);
    bool removePipeline(int channel_id);
    PipelineStats getStats(int channel_id) const;
    std::vector<int> getAllChannels() const;
    
private:
    std::unique_ptr<VideoPipelineManager> pipeline_mgr_;
    bool running_ = false;
    boost::asio::io_context io_ctx_;
    std::thread io_thread_;
};

} // namespace services
} // namespace web

#### 步骤 2: 实现服务

文件: `modules/web/src/services/video_pipeline_service.cpp`

```cpp
#include "web/services/video_pipeline_service.h"
#include "log/logmanager.h"
#include <boost/asio.hpp>

namespace web {
namespace services {

VideoPipelineService::VideoPipelineService() {
}

VideoPipelineService::~VideoPipelineService() {
    stop();
}

bool VideoPipelineService::initialize() {
    LOG_MAIN_INFO_AT("[VideoPipelineService] Initializing...");
    
    try {
        pipeline_mgr_ = std::make_unique<VideoPipelineManager>();
        LOG_MAIN_INFO_AT("[VideoPipelineService] Initialized successfully");
        return true;
    } catch (const std::exception& e) {
        LOG_MAIN_ERROR_AT("[VideoPipelineService] Initialization failed: {}", e.what());
        return false;
    }
}

bool VideoPipelineService::start() {
    if (running_) {
        LOG_MAIN_WARN_AT("[VideoPipelineService] Already running");
        return true;
    }
    
    LOG_MAIN_INFO_AT("[VideoPipelineService] Starting...");
    
    try {
        // 启动 io_context 线程
        io_thread_ = std::thread([this]() {
            boost::asio::executor_work_guard<boost::asio::io_context::executor_type> work(io_ctx_.get_executor());
            io_ctx_.run();
        });
        
        running_ = true;
        LOG_MAIN_INFO_AT("[VideoPipelineService] Started successfully");
        return true;
    } catch (const std::exception& e) {
        LOG_MAIN_ERROR_AT("[VideoPipelineService] Start failed: {}", e.what());
        return false;
    }
}

void VideoPipelineService::stop() {
    if (!running_) {
        return;
    }
    
    LOG_MAIN_INFO_AT("[VideoPipelineService] Stopping...");
    
    // 停止所有流水线
    if (pipeline_mgr_) {
        pipeline_mgr_->stopAll();
    }
    
    // 停止 io_context
    io_ctx_.stop();
    if (io_thread_.joinable()) {
        io_thread_.join();
    }
    
    running_ = false;
    LOG_MAIN_INFO_AT("[VideoPipelineService] Stopped");
}

// ... 实现其他业务方法

} // namespace services
} // namespace web
```

#### 步骤 3: 注册服务

在 `main.cpp` 中使用：

```cpp
#include "web/core/service_manager.h"
#include "web/services/video_pipeline_service.h"

int main() {
    // 创建服务管理器
    auto& manager = web::core::ServiceManager::getInstance();
    
    // 注册视频流水线服务
    auto video_service = std::make_shared<web::services::VideoPipelineService>();
    manager.registerService(video_service);
    
    // 初始化并启动所有服务
    if (!manager.initializeAll()) {
        LOG_MAIN_ERROR_AT("Failed to initialize services");
        return 1;
    }
    
    if (!manager.startAll()) {
        LOG_MAIN_ERROR_AT("Failed to start services");
        return 1;
    }
    
    LOG_MAIN_INFO_AT("All services started successfully!");
    
    // 等待信号...
    
    // Graceful shutdown
    manager.stopAll();
    LOG_MAIN_INFO_AT("Application stopped");
    
    return 0;
}
```

## 🐛 常见问题

### Q1: 循环依赖怎么办？
**A**: 使用前向声明 + 接口抽象，避免直接依赖具体实现。

### Q2: 服务启动顺序如何确定？
**A**: 在 ServiceManager 中维护 `startup_order_` 列表，按依赖关系排序。

### Q3: 如何处理服务间的通信？
**A**: 通过 ServiceManager 获取服务实例，调用其公开方法。

### Q4: 测试时如何 mock 服务？
**A**: 实现 IService 接口的 Mock 类，注入到 ServiceManager。
